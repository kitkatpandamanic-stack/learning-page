The Projects API works. But "it worked when I tried it" isn't good enough for an API other apps depend on: the next change could quietly break an endpoint you didn't touch. In this final part you'll write a **full API test suite**, so every endpoint, every error and every save is checked automatically in under a second. 🧪

You'll use everything from [Fixtures and Parametrize](/learn/python/fixtures-and-parametrize) and the `client` fixture with `dependency_overrides` from the [To-do API](/learn/python/todo-api), now with a real database underneath.

## Two layers, two kinds of tests

We already have **repository tests** from Part 1. They're fast and precise, but they can't see anything HTTP-related. **API tests** send real requests through `TestClient` and check what a client would see. Together they cover different bugs:

| A bug in... | Caught by |
| --- | --- |
| a query, a filter, the cascade | repository tests (and API tests) |
| a `Field` rule, a missing 404, a wrong status code | API tests |
| the JSON shape (`response_model`, `from_attributes`) | API tests |
| wiring: `Depends`, a forgotten `commit()` in an endpoint | API tests |

Most of an API's behaviour is in the second group, so a good suite has **lots** of API tests. They're still fast, because the database is in memory.

## A fresh database for every test

API tests must never use the app's real database: they'd fill it with test data, and tests would see each other's leftovers and pass or fail depending on their order. The fix is the trick from the To-do API: a `client` fixture that **overrides `get_session`**, so that during the test every endpoint gets a session on a **brand-new, empty test database**. Here's the whole pattern on a small notes API:

<CodeExample output={`...........                                          [100%]
11 passed in 0.01s`}>

```python
import pytest
from fastapi import Depends, FastAPI, HTTPException
from fastapi.testclient import TestClient
from pydantic import BaseModel, ConfigDict, Field
from sqlalchemy import create_engine, select
from sqlalchemy.orm import (
    DeclarativeBase, Mapped, Session, mapped_column,
)
from sqlalchemy.pool import StaticPool

class Base(DeclarativeBase):
    pass

class Note(Base):
    __tablename__ = "notes"
    id: Mapped[int] = mapped_column(primary_key=True)
    text: Mapped[str]

class NoteIn(BaseModel):
    text: str = Field(min_length=1)

class NoteOut(BaseModel):
    model_config = ConfigDict(from_attributes=True)
    id: int
    text: str

engine = create_engine(
    "sqlite://",
    connect_args={"check_same_thread": False},
    poolclass=StaticPool,
)
Base.metadata.create_all(engine)

def get_session():
    with Session(engine) as session:
        yield session

def get_note_or_404(
    note_id: int, session: Session = Depends(get_session)
) -> Note:
    note = session.get(Note, note_id)
    if note is None:
        raise HTTPException(status_code=404, detail="Note not found")
    return note

app = FastAPI()

@app.post("/notes", status_code=201, response_model=NoteOut)
def api_add_note(
    data: NoteIn, session: Session = Depends(get_session)
):
    note = Note(text=data.text)
    session.add(note)
    session.commit()
    return note

@app.get("/notes/{note_id}", response_model=NoteOut)
def api_get_note(note: Note = Depends(get_note_or_404)):
    return note

@app.delete("/notes/{note_id}", status_code=204)
def api_delete_note(
    note: Note = Depends(get_note_or_404),
    session: Session = Depends(get_session),
):
    session.delete(note)
    session.commit()

# ---------- tests ----------

@pytest.fixture
def client():
    test_engine = create_engine(
        "sqlite://",
        connect_args={"check_same_thread": False},
        poolclass=StaticPool,
    )
    Base.metadata.create_all(test_engine)

    def get_test_session():
        with Session(test_engine) as session:
            yield session

    app.dependency_overrides[get_session] = get_test_session
    yield TestClient(app)
    app.dependency_overrides.clear()

@pytest.fixture
def note(client):
    return client.post("/notes", json={"text": "Buy milk"}).json()

def test_add_note(client):
    response = client.post("/notes", json={"text": "Buy milk"})
    assert response.status_code == 201
    assert response.json() == {"id": 1, "text": "Buy milk"}

def test_get_note(client, note):
    assert client.get(f"/notes/{note['id']}").json() == note

def test_every_test_has_its_own_database(client, note):
    assert note["id"] == 1

def test_the_app_database_is_never_touched(client, note):
    with Session(engine) as session:
        assert session.scalars(select(Note)).all() == []

@pytest.mark.parametrize("method, url", [
    ("GET", "/notes/99"),
    ("DELETE", "/notes/99"),
    ("GET", "/notes/0"),
])
def test_unknown_ids_give_404(client, method, url):
    response = client.request(method, url)
    assert response.status_code == 404
    assert response.json() == {"detail": "Note not found"}

@pytest.mark.parametrize("body", [
    {},
    {"text": ""},
    {"text": None},
    {"txt": "Buy milk"},
])
def test_bad_data_gives_422(client, body):
    response = client.post("/notes", json=body)
    assert response.status_code == 422

pytest.main([__file__])
```

</CodeExample>

How the `client` fixture works, step by step:

1. It creates a **new engine** for this one test: an empty in-memory database, with `StaticPool` and `check_same_thread=False` for the same reasons as in Part 2.
2. `create_all` adds the tables.
3. `get_test_session` is a dependency with `yield`, just like `get_session`, but on the test engine.
4. `app.dependency_overrides[get_session] = get_test_session` makes **every** endpoint and dependency that asks for `get_session` (including `get_note_or_404`) get the test session instead.
5. After `yield`, the overrides are cleared, so nothing leaks into the next test.

The last two tests prove it: each test's first note gets id `1`, and the app's own database stays empty. In real projects you sometimes see a session-scoped engine with a transaction that's rolled back after each test, which is faster for big databases. For in-memory SQLite, a new engine per test is simple and quick.

## Fixtures on fixtures

`note` is a fixture that **asks for another fixture**, `client`. pytest sees that, creates the client (and its fresh database) first, then the note. Tests that need existing data just list both: `def test_get_note(client, note)`. Both get the **same** client, because pytest creates each fixture once per test.

In the Projects API, a `project` fixture does the same job: it creates a project called `"Garden"` through the API and returns its JSON, so task tests can start with `def test_...(client, project)`.

## Testing errors with parametrize

Errors are behaviour too, and APIs have a lot of them: every endpoint with an id can answer 404, and every endpoint with a body can answer 422. `@pytest.mark.parametrize` turns that into a table of cases instead of a pile of copied tests:

- `client.request(method, url)` sends any HTTP method, so one test can try `GET`, `DELETE` and more. You can also pass a body: `client.request("PATCH", url, json={...})`.
- Each case shows up as its own test, so a failure names exactly which request broke, like `test_unknown_ids_give_404[DELETE-/notes/99]`.
- The 422 cases list the classic mistakes: a missing field, an empty value, `None`, and a misspelt key.

<Callout type="tip" title="What pros check in API tests">

For every endpoint: the **status code**, the **whole JSON body** (comparing with `==` against a full dict catches extra or missing fields) and the **side effect** (is it really saved, really deleted?). Then the error cases. That checklist catches most real-world API bugs.

</Callout>

## Did it really save?

Here's a bug the response alone can't reveal. This `PATCH` endpoint changes the note but forgets to commit:

<CodeExample output={`PATCH says: {'id': 1, 'text': 'Buy oat milk'}
GET says:   {'id': 1, 'text': 'Buy milk'}`}>

</CodeExample>

The `PATCH` response looks perfect, because it's built from the changed object while the session is still open. Then the request ends, the session closes, the change is rolled back, and the next request sees the old text. A test that only checks the `PATCH` response would pass. So after every write, **read it back in a new request**: `PATCH`, then `GET`; `DELETE`, then `GET` and expect 404.

Our repository test from Part 1 can't catch this either: it checks the same `task` object in the same session. That's exactly why both kinds of tests are worth having.

## Running it for real

On your computer, the suite is split into files, and the `client` fixture goes in **`tests/conftest.py`** so every test file can use it:

```python
# tests/conftest.py
import pytest
from fastapi.testclient import TestClient
from sqlalchemy import create_engine
from sqlalchemy.orm import Session
from sqlalchemy.pool import StaticPool

from app.database import get_session
from app.main import app
from app.models import Base


@pytest.fixture
def client():
    test_engine = create_engine(
        "sqlite://",
        connect_args={"check_same_thread": False},
        poolclass=StaticPool,
    )
    Base.metadata.create_all(test_engine)

    def get_test_session():
        with Session(test_engine) as session:
            yield session

    app.dependency_overrides[get_session] = get_test_session
    yield TestClient(app)
    app.dependency_overrides.clear()
```

Then, from the project folder (with an empty `__init__.py` in `app/` and `tests/`):

```bash
pip install "fastapi[standard]" sqlalchemy pytest pytest-cov
pytest
pytest --cov=app --cov-report=term-missing
fastapi dev app/main.py
```

`pytest` finds and runs every `test_*.py` file. The coverage report from [Mocks and Coverage](/learn/python/mocks-and-coverage) shows any lines your tests never reached: with the suite you're about to write, the endpoints and repository should be at or near **100%**. And `fastapi dev` starts the server so you can click through `/docs`.

# ---------- models ----------

class Base(DeclarativeBase):
    pass


class Project(Base):
    __tablename__ = "projects"

    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str] = mapped_column(String(100))
    tasks: Mapped[list["Task"]] = relationship(
        back_populates="project", cascade="all, delete-orphan"
    )


class Task(Base):
    __tablename__ = "tasks"

    id: Mapped[int] = mapped_column(primary_key=True)
    title: Mapped[str] = mapped_column(String(200))
    done: Mapped[bool] = mapped_column(default=False)
    project_id: Mapped[int] = mapped_column(ForeignKey("projects.id"))
    project: Mapped["Project"] = relationship(back_populates="tasks")


# ---------- repository ----------

def create_project(session: Session, name: str) -> Project:
    project = Project(name=name)
    session.add(project)
    session.commit()
    return project


def get_project(session: Session, project_id: int) -> Project | None:
    return session.get(Project, project_id)


def list_projects(session: Session) -> list[Project]:
    return list(session.scalars(select(Project).order_by(Project.id)))


def delete_project(session: Session, project: Project) -> None:
    session.delete(project)
    session.commit()


def add_task(session: Session, project: Project, title: str) -> Task:
    task = Task(title=title, project=project)
    session.add(task)
    session.commit()
    return task


def get_task(session: Session, task_id: int) -> Task | None:
    return session.get(Task, task_id)


def list_tasks(
    session: Session, project_id: int, done: bool | None = None
) -> list[Task]:
    stmt = select(Task).where(Task.project_id == project_id)
    if done is not None:
        stmt = stmt.where(Task.done == done)
    return list(session.scalars(stmt.order_by(Task.id)))


def update_task(session: Session, task: Task, changes: dict) -> Task:
    for field, value in changes.items():
        setattr(task, field, value)
    session.commit()
    return task


def delete_task(session: Session, task: Task) -> None:
    session.delete(task)
    session.commit()
